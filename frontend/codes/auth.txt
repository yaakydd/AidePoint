// navigation/AppNavigator.js
//
// The root navigator. Decides which screen to show based on auth state.
//
// FLOW:
//   isLoading = true           → SplashScreen (checking saved session)
//   user = null                → AuthNavigator (login / signup / onboarding)
//   user + needsConsent = true → ConsentScreen (image storage preference)
//   user + needsConsent = false→ MainAppNavigator (the main tabbed app)

import React from 'react';
import { View, ActivityIndicator, StyleSheet } from 'react-native';
import { NavigationContainer } from '@react-navigation/native';
import { createNativeStackNavigator } from '@react-navigation/native-stack';
import { createBottomTabNavigator } from '@react-navigation/bottom-tabs';
import { Ionicons } from '@expo/vector-icons';
import { Platform } from 'react-native';

import { useAuth } from '../context/AuthContext';
import { COLORS, FONTS, layout } from '../assets/theme';

// ── Auth screens ──────────────────────────────────────────────────────────────
import OnboardingScreen from '../screens/auth/OnboardingScreen';
import UserTypeScreen   from '../screens/auth/UserTypeScreen';
import SignInScreen     from '../screens/auth/SignIn';
import SignUpScreen     from '../screens/auth/SignUp';
import ConsentScreen    from '../screens/auth/ConsentScreen';

// ── Main app screens ──────────────────────────────────────────────────────────
import HomeScreen    from '../screens/HomeScreen';
import ScanScreen    from '../screens/ScanScreen';
import ReportsScreen from '../screens/ReportsScreen';
import Chatbot       from '../screens/Chatbot';
import ProfileScreen from '../screens/ProfileScreen';

// ─── STACK / TAB INSTANCES ───────────────────────────────────────────────────
const AuthStack = createNativeStackNavigator();
const AppStack  = createNativeStackNavigator();
const Tab       = createBottomTabNavigator();
const ScanStack = createNativeStackNavigator();


// ─── LOADING SPLASH ───────────────────────────────────────────────────────────
// Shown for a split second on startup while we check the saved session.
function SplashScreen() {
  return (
    <View style={styles.splash}>
      <ActivityIndicator size="large" color={COLORS.primary} />
    </View>
  );
}


// ─── AUTH NAVIGATOR ───────────────────────────────────────────────────────────
// Shown when user is null (not logged in).
// Includes: Onboarding → UserType → SignUp / SignIn
function AuthNavigator() {
  const { isFirstLaunch } = useAuth();

  return (
    <AuthStack.Navigator screenOptions={{ headerShown: false }}>
      {/* Show onboarding slides only on the very first launch ever */}
      {isFirstLaunch && (
        <AuthStack.Screen name="Onboarding" component={OnboardingScreen} />
      )}
      <AuthStack.Screen name="UserType" component={UserTypeScreen} />
      <AuthStack.Screen name="SignIn"   component={SignInScreen} />
      <AuthStack.Screen name="SignUp"   component={SignUpScreen} />
    </AuthStack.Navigator>
  );
}


// ─── SCAN STACK NAVIGATOR ─────────────────────────────────────────────────────
// The Scan tab has its own sub-stack so CameraView can be pushed on top.
function ScanStackNavigator() {
  return (
    <ScanStack.Navigator screenOptions={{ headerShown: false }}>
      <ScanStack.Screen name="ScanHome"   component={ScanScreen} />
      {/* CameraView is pushed when user taps "Take Photo" */}
      {/* <ScanStack.Screen name="CameraView" component={CameraView} /> */}
    </ScanStack.Navigator>
  );
}


// ─── MAIN APP NAVIGATOR (BOTTOM TABS) ─────────────────────────────────────────
// Shown when user is logged in and has completed consent.
function MainAppNavigator() {
  return (
    <Tab.Navigator
      screenOptions={({ route }) => ({
        headerShown: false,
        tabBarActiveTintColor:   COLORS.primary,
        tabBarInactiveTintColor: COLORS.textSecondary,
        tabBarStyle: {
          backgroundColor: COLORS.surface,
          borderTopColor:  COLORS.border,
          height: Platform.OS === 'ios' ? 83 : 60,
          paddingBottom: Platform.OS === 'ios' ? layout.bottomInset : 8,
          paddingTop: 8,
        },
        tabBarLabelStyle: {
          fontSize: FONTS.xs,
          fontWeight: FONTS.medium,
        },
        tabBarIcon: ({ focused, color, size }) => {
          const icons = {
            Home:    focused ? 'home'          : 'home-outline',
            Scan:    focused ? 'scan-circle'   : 'scan-circle-outline',
            Reports: focused ? 'document-text' : 'document-text-outline',
            Chatbot: focused ? 'chatbubbles'   : 'chatbubbles-outline',
            Profile: focused ? 'person-circle' : 'person-circle-outline',
          };
          return (
            <Ionicons
              name={icons[route.name] ?? 'ellipse'}
              size={size}
              color={color}
            />
          );
        },
      })}
    >
      <Tab.Screen name="Home"    component={HomeScreen} />
      <Tab.Screen name="Scan"    component={ScanStackNavigator} />
      <Tab.Screen name="Reports" component={ReportsScreen} />
      <Tab.Screen name="Chatbot" component={Chatbot} />
      <Tab.Screen name="Profile" component={ProfileScreen} />
    </Tab.Navigator>
  );
}


// ─── ROOT NAVIGATOR ───────────────────────────────────────────────────────────
// This is the top-level component exported and used in App.js.
export default function AppNavigator() {
  const { user, isLoading, needsConsent } = useAuth();

  return (
    <NavigationContainer>
      {/* Case 1: Still checking session from storage — show spinner */}
      {isLoading ? (
        <SplashScreen />
      ) : (
        <AppStack.Navigator screenOptions={{ headerShown: false }}>

          {/* Case 2: No user — show auth flow */}
          {!user ? (
            <AppStack.Screen name="Auth" component={AuthNavigator} />

          // Case 3: User just registered — show image consent screen
          ) : needsConsent ? (
            <AppStack.Screen name="Consent" component={ConsentScreen} />

          // Case 4: Fully authenticated — show main app
          ) : (
            <AppStack.Screen name="Main" component={MainAppNavigator} />
          )}

        </AppStack.Navigator>
      )}
    </NavigationContainer>
  );
}


const styles = StyleSheet.create({
  splash: {
    flex: 1,
    justifyContent: 'center',
    alignItems: 'center',
    backgroundColor: COLORS.background,
  },
});













// context/AuthContext.js
//
// Central auth state for the entire app.
// Wrap your app root in <AuthProvider> and call useAuth() in any screen.

import React, { createContext, useContext, useState, useEffect } from 'react';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { supabase } from '../utils/supabase';

const STORAGE_KEYS = {
  HAS_LAUNCHED: 'aidepoint_has_launched',
};

const AuthContext = createContext({});

export function AuthProvider({ children }) {

  const [user,          setUser]          = useState(null);
  const [isLoading,     setIsLoading]     = useState(true);
  const [isFirstLaunch, setIsFirstLaunch] = useState(false);
  const [authError,     setAuthError]     = useState(null);

  // ── NEW: controls whether ConsentScreen shows after signup ───────────────
  // When register() succeeds, this becomes true.
  // When completeOnboarding() is called, this becomes false
  // and AppNavigator switches to the main app.
  const [needsConsent, setNeedsConsent] = useState(false);


  // ── Startup: check session + first launch ─────────────────────────────────
  useEffect(() => {
    checkFirstLaunch();

    // Check if a session already exists on this device (from a previous login).
    // @supabase/supabase-js reads it from AsyncStorage automatically.
    supabase.auth.getSession().then(({ data: { session } }) => {
      if (session) {
        fetchAndSetUser(session);
      } else {
        setIsLoading(false);
      }
    });

    // This listener fires on: login, logout, token refresh.
    // It is the single source of truth for auth state.
    const { data: { subscription } } = supabase.auth.onAuthStateChange(
      async (_event, session) => {
        if (session) {
          await fetchAndSetUser(session);
        } else {
          setUser(null);
          setIsLoading(false);
        }
      }
    );

    return () => subscription.unsubscribe();
  }, []);


  // ── Helpers ───────────────────────────────────────────────────────────────

  async function checkFirstLaunch() {
    try {
      const seen = await AsyncStorage.getItem(STORAGE_KEYS.HAS_LAUNCHED);
      if (!seen) {
        setIsFirstLaunch(true);
        await AsyncStorage.setItem(STORAGE_KEYS.HAS_LAUNCHED, 'true');
      }
    } catch (e) {
      console.warn('AuthContext: checkFirstLaunch error', e);
    }
  }

  // Fetches the user's profile row from Supabase and builds the user object
  // that every screen in the app reads via useAuth().
  async function fetchAndSetUser(session) {
    try {
      const { data: profile, error } = await supabase
        .from('profiles')
        .select('*')
        .eq('id', session.user.id)
        .single();

      if (error || !profile) {
        // Rare race condition: trigger hasn't created the profile row yet.
        // Fall back to auth metadata.
        const meta = session.user.user_metadata ?? {};
        setUser({
          id:          session.user.id,
          name:        meta.name         ?? 'Unknown',
          email:       session.user.email,
          role:        meta.role         ?? 'lab_technician',
          userType:    meta.user_type    ?? 'solo',
          hospitalId:  meta.hospital_id  ?? null,
          storeImages: false,
          token:       session.access_token,
        });
      } else {
        setUser({
          id:          session.user.id,
          name:        profile.name,
          email:       session.user.email,
          role:        profile.role,
          userType:    profile.user_type,
          hospitalId:  profile.hospital_id ?? null,
          storeImages: profile.store_images ?? false,
          token:       session.access_token,
        });
      }
    } catch (e) {
      console.error('AuthContext: fetchAndSetUser error', e);
    } finally {
      setIsLoading(false);
    }
  }


  // ── Auth Functions ─────────────────────────────────────────────────────────

  // LOGIN
  // Called from SignInScreen.
  // On success, onAuthStateChange fires automatically and sets the user.
  // Returns { success: true } or { success: false, error: 'message' }
  async function login(email, password) {
    setAuthError(null);
    try {
      const { error } = await supabase.auth.signInWithPassword({
        email: email.trim().toLowerCase(),
        password,
      });
      if (error) throw error;
      return { success: true };
    } catch (e) {
      const msg = e.message ?? 'Login failed. Please try again.';
      setAuthError(msg);
      return { success: false, error: msg };
    }
  }


  // REGISTER
  // Called from SignUpScreen.
  //
  // Accepts: { name, email, password, userType }
  //   name     → user's full name
  //   email    → their email address
  //   password → chosen password (min 8 chars)
  //   userType → 'solo' or 'hospital'
  //
  // What happens behind the scenes:
  //   1. supabase.auth.signUp() creates a new row in auth.users
  //   2. The on_auth_user_created trigger fires and creates a profiles row
  //   3. onAuthStateChange fires SIGNED_IN — user is set in state
  //   4. needsConsent is set to true so ConsentScreen is shown BEFORE main app
  //
  // Returns { success: true } or { success: false, error: 'message' }
  async function register({ name, email, password, userType = 'solo' }) {
    setAuthError(null);
    try {
      const { data, error } = await supabase.auth.signUp({
        email: email.trim().toLowerCase(),
        password,
        options: {
          // This metadata is read by the on_auth_user_created trigger
          // to populate the profiles table row automatically.
          data: {
            name:      name.trim(),
            role:      'lab_technician',
            user_type: userType,
            // hospital_id is null for solo — the SSO flow handles hospital users
          },
        },
      });

      if (error) throw error;

      // requiresConfirmation is true when Supabase email confirmation is ON.
      // During development we turn it OFF, but we handle both cases.
      const requiresConfirmation = !data.session;

      if (!requiresConfirmation) {
        // Session exists immediately (email confirmation is OFF).
        // onAuthStateChange will fire and set the user.
        // We flag needsConsent so ConsentScreen is shown before main app.
        setNeedsConsent(true);
      }

      return { success: true, requiresConfirmation };

    } catch (e) {
      const msg = e.message ?? 'Registration failed. Please try again.';
      setAuthError(msg);
      return { success: false, error: msg };
    }
  }


  // LOGOUT
  async function logout() {
    setAuthError(null);
    setNeedsConsent(false);
    await supabase.auth.signOut();
    // onAuthStateChange fires and sets user to null.
  }


  // COMPLETE ONBOARDING (called from ConsentScreen)
  //
  // Saves the user's image storage consent to their profile in the database.
  // Then clears needsConsent so AppNavigator switches to the main app.
  //
  // storeImages: boolean — did the user agree to store images?
  async function completeOnboarding(storeImages) {
    try {
      if (!user) return { success: false };

      const { error } = await supabase
        .from('profiles')
        .update({ store_images: storeImages })
        .eq('id', user.id);

      if (error) {
        console.warn('AuthContext: completeOnboarding DB error', error);
        // Even if DB fails, still let the user into the app.
        // Default (false) will be used until they change it in Profile.
      }

      // Update local user state immediately.
      setUser(prev => ({ ...prev, storeImages }));

      // This is the key line — clears the flag so AppNavigator
      // stops showing ConsentScreen and shows the main app instead.
      setNeedsConsent(false);

      return { success: true };
    } catch (e) {
      console.error('AuthContext: completeOnboarding error', e);
      setNeedsConsent(false); // Always let the user through
      return { success: false };
    }
  }


  // UPDATE STORE IMAGES (called from ProfileScreen)
  //
  // Lets the user change their image storage preference after onboarding.
  async function updateStoreImages(storeImages) {
    try {
      if (!user) return { success: false };

      const { error } = await supabase
        .from('profiles')
        .update({ store_images: storeImages })
        .eq('id', user.id);

      if (error) throw error;

      setUser(prev => ({ ...prev, storeImages }));
      return { success: true };
    } catch (e) {
      const msg = e.message ?? 'Could not update preference.';
      return { success: false, error: msg };
    }
  }


  // CLEAR ERROR
  function clearError() {
    setAuthError(null);
  }


  return (
    <AuthContext.Provider value={{
      user,
      isLoading,
      isFirstLaunch,
      authError,
      needsConsent,
      login,
      register,
      logout,
      clearError,
      completeOnboarding,
      updateStoreImages,
    }}>
      {children}
    </AuthContext.Provider>
  );
}

export const useAuth = () => useContext(AuthContext);
